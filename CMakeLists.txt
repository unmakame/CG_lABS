cmake_minimum_required(VERSION 3.25)
project(CG_All_Labs LANGUAGES C CXX)
if(NOT DEFINED ENV{VULKAN_SDK})
  message(FATAL_ERROR "Set VULKAN_SDK to ~/VulkanSDK/1.4.321.0/macOS before configuring")
endif()
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glfw build-glfw EXCLUDE_FROM_ALL)
foreach(LAB RANGE 1 3)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lab${LAB}/CMakeLists.txt")
    add_subdirectory(lab${LAB} build-lab${LAB})
  endif()
endforeach()
